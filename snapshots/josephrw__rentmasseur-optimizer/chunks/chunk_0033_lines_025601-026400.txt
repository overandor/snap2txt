    def cycle(self):
        """One daemon cycle."""
        now = time.time()

        if now - self.last_visibility_check >= VISIBILITY_CHECK_INTERVAL:
            self.ensure_visible()
            self.last_visibility_check = now

        if now - self.last_avail_check >= AVAIL_CHECK_INTERVAL:
            self.ensure_available()
            self.last_avail_check = now

        if now - self.last_stats_check >= STATS_INTERVAL:
            self.collect_stats()
            self.check_search_position()
            self.last_stats_check = now

        if now - self.last_bio_check >= BIO_EXPERIMENT_INTERVAL:
            self.bio_experiment(auto_apply=False)
            self.last_bio_check = now

        if now - self.last_dashboard_print >= DASHBOARD_INTERVAL:
            self.print_dashboard()
            self.last_dashboard_print = now

    def run_once(self):
        if not self.login():
            return
        self.print_dashboard()
        self.ensure_visible()
        self.ensure_available()
        self.collect_stats()
        self.check_search_position()
        self.bio_experiment(auto_apply=False)
        draft = self.draft_blog_post()
        print(f"\nBlog draft (not published):\n{draft['title']}\n{draft['body']}\n")

    def run_daemon(self):
        log.info("=== ProfileOps daemon starting ===")
        if not self.login():
            log.error("Initial login failed, retrying in 60s")
            time.sleep(60)
            if not self.login():
                log.error("Login failed twice, exiting")
                return

        self.print_dashboard()
        while self.running:
            try:
                self.cycle()
            except KeyboardInterrupt:
                log.info("Shutdown requested")
                self.running = False
                break
            except Exception as e:
                log.error("Cycle error: %s", e)
                receipt("daemon_cycle", False, str(e))
                time.sleep(30)
            time.sleep(60)
        log.info("ProfileOps daemon stopped")

std::string json_unescape(const std::string& s) {
    std::string out;
    for (size_t i = 0; i < s.size(); i++) {
        if (s[i] == '\\' && i + 1 < s.size()) {
            char next = s[i+1];
            if (next == 'n') out += '\n';
            else if (next == 't') out += '\t';
            else if (next == 'r') out += '\r';
            else out += next;
            i++;
        } else if (s[i] != '"') {
            out += s[i];
        }
    }
    return out;
}

std::string extract_json_string(const std::string& line, const std::string& key) {
    size_t pos = line.find("\"" + key + "\":");
    if (pos == std::string::npos) return "";
    pos = line.find("\"", pos + key.size() + 3);
    if (pos == std::string::npos) return "";
    size_t end = pos + 1;
    while (end < line.size() && (line[end] != '"' || line[end-1] == '\\')) end++;
    return json_unescape(line.substr(pos + 1, end - pos - 1));
}

std::string to_lower(const std::string& s) {
    std::string r = s;

std::vector<std::string> tokenize(const std::string& text) {
    std::vector<std::string> tokens;
    std::string cur;
    for (char c : text) {
        if (std::isalpha(static_cast<unsigned char>(c)) || c == '\'') {
            cur += std::tolower(c);
        } else if (!cur.empty()) {
            tokens.push_back(cur);
            cur.clear();
        }
    }
    if (!cur.empty()) tokens.push_back(cur);
    return tokens;
}

std::vector<std::string> split_sentences(const std::string& text) {
    std::vector<std::string> sents;
    std::string cur;
    for (char c : text) {
        cur += c;
        if (c == '.' || c == '!' || c == '?') {
            if (!cur.empty()) {
                sents.push_back(cur);
                cur.clear();
            }
        }
    }
    if (!cur.empty()) sents.push_back(cur);
    return sents;
}
