def _groq_chat(system_prompt: str, user_prompt: str, max_tokens: int = 800, retries: int = 3) -> Optional[str]:
    if not GROQ_API_KEY:
        logger.error("No GROQ_API_KEY")
        return None
    for attempt in range(retries):
        try:
            resp = requests.post(
                "https://api.groq.com/openai/v1/chat/completions",
                headers={"Authorization": f"Bearer {GROQ_API_KEY}", "Content-Type": "application/json"},
                json={
                    "model": GROQ_MODEL,
                    "messages": [
                        {"role": "system", "content": system_prompt},
                        {"role": "user", "content": user_prompt},
                    ],
                    "temperature": 0.9,
                    "max_tokens": max_tokens,
                },
                timeout=90,
            )
            if resp.status_code == 429:
                wait = 5 * (attempt + 1)
                logger.warning("Groq rate limited, waiting %ds", wait)
                time.sleep(wait)
                continue
            resp.raise_for_status()
            return resp.json()["choices"][0]["message"]["content"].strip()
        except Exception as e:
            logger.error("Groq API error: %s", e)
            if attempt < retries - 1:
                time.sleep(5 * (attempt + 1))
    return None


def _groq_json(system_prompt: str, user_prompt: str, max_tokens: int = 800) -> Optional[dict]:
    raw = _groq_chat(system_prompt, user_prompt, max_tokens=max_tokens)
    if not raw:
        return None
    try:
        if raw.startswith("```"):
            raw = raw.split("```")[1]
            if raw.startswith("json"):
                raw = raw[4:]
        return json.loads(raw)
    except json.JSONDecodeError:
        logger.warning("Failed to parse Groq JSON: %s", raw[:200])
        return None

        time.sleep(1)

    parser.add_argument("--dry-run", action="store_true", help="Don't upload, just test")
    parser.add_argument("--competitors-only", action="store_true", help="Only scrape competitors")
    parser.add_argument("--min-score", type=int, default=75, help="Minimum score to upload")
    args = parser.parse_args()

    os.makedirs(BIOS_DIR, exist_ok=True)

    competitor_bios = scrape_competitor_bios()

        return

    if not variants:

    # Step 3: A/B test
    winner = ab_test_bios(variants, competitor_bios)
    if not winner:
        logger.error("No winner determined")
        sys.exit(1)

    winner_score = winner["score"].get("overall_score", 0)
    if winner_score < args.min_score:
        logger.warning("Winner score %d below minimum %d — not uploading", winner_score, args.min_score)
        print(json.dumps({"winner": winner["strategy"], "score": winner_score, "uploaded": False, "reason": "below_min_score"}, indent=2))
        return

    # Step 4: Verify against current bio

    if current_bio and not verify_bio_beats_current(winner["bio"], current_bio):
        logger.warning("Winner did not beat current bio — keeping current")
        print(json.dumps({"winner": winner["strategy"], "score": winner_score, "uploaded": False, "reason": "current_bio_is_better"}, indent=2))
        return

        logger.info("Dry run — not uploading")
        print(json.dumps({"winner": winner["strategy"], "score": winner_score, "uploaded": False, "reason": "dry_run"}, indent=2))
        return

    success = upload_bio(winner["bio"])
    print(json.dumps({
        "winner": winner["strategy"],
        "score": winner_score,
        "uploaded": success,
        "competitors_analyzed": len(competitor_bios),
        "variants_tested": len(variants),
        "date": datetime.now(timezone.utc).isoformat(),
    }, indent=2))
    sys.exit(0 if success else 1)

def organic_visit_delay():
    """Vary delays to mimic human browsing patterns — not robotic fixed intervals."""
    # Sometimes quick, sometimes slow, occasionally pause like reading
    roll = random.random()
    if roll < 0.6:
        # Normal browsing: 2-5s
        time.sleep(random.uniform(2.0, 5.0))
    elif roll < 0.85:
        # Reading profile: 5-12s
        time.sleep(random.uniform(5.0, 12.0))
    else:
        # Distracted / long pause: 10-25s
        time.sleep(random.uniform(10.0, 25.0))


def organic_batch_pause(batch_num: int):
    """Every N visits, take a longer break like a human stepping away."""
    if batch_num > 0 and batch_num % random.choice([15, 20, 25]) == 0:
        pause = random.uniform(30, 90)
        log(f"  Organic batch pause: {pause:.0f}s (after {batch_num} visits)")
        time.sleep(pause)


def organic_shuffle(visitors: list[dict]) -> list[dict]:
    """Shuffle visit order — never visit in the same sequence twice."""
    shuffled = visitors.copy()
    random.shuffle(shuffled)
    return shuffled


def organic_session_size(total: int) -> int:
    """Don't visit all 450 in one robotic batch — visit a natural subset per session."""
    if total <= 20:
        return total
    # Visit 30-70% of remaining, leaving organic gaps
    return max(20, int(total * random.uniform(0.3, 0.7)))


def human_type(element, text: str, min_delay: float = 0.05, max_delay: float = 0.15):
    for char in text:
        element.send_keys(char)
        time.sleep(random.uniform(min_delay, max_delay))

# ─── Screenshot Compression ───────────────────────────────────────────

def compress_screenshot(png_path: Path, max_width: int = 400, quality: int = 50) -> tuple[str, int]:
    """Compress a PNG screenshot to a small JPEG thumbnail, return base64 + size in KB.
    Original PNG is deleted after compression."""
    try:
        from PIL import Image
        img = Image.open(str(png_path))
        ratio = max_width / img.width
        new_size = (max_width, int(img.height * ratio))
        img = img.resize(new_size, Image.LANCZOS)
        img = img.convert("RGB")
        buf = io.BytesIO()
        img.save(buf, format="JPEG", quality=quality, optimize=True)
        thumb_b64 = base64.b64encode(buf.getvalue()).decode("ascii")
        thumb_kb = round(len(buf.getvalue()) / 1024, 1)
        # Save thumbnail file
        thumb_path = THUMBNAILS / (png_path.stem + ".jpg")
        thumb_path.write_bytes(buf.getvalue())
        # Delete original PNG
        png_path.unlink(missing_ok=True)
        return thumb_b64, thumb_kb
    except Exception as e:
        log(f"Screenshot compression failed: {e}", "WARN")
        return "", 0


def already_visited_today(username: str) -> bool:
    """Check if we already visited this profile today."""
    conn = sqlite3.connect(str(DB_PATH))
    today = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    row = conn.execute(
        "SELECT visited_at FROM visit_log WHERE username=? AND status='ok' ORDER BY visited_at DESC LIMIT 1",
        (username,)
    ).fetchone()
    conn.close()
    if not row:
        return False
    try:
        last_visit = row[0][:10]  # YYYY-MM-DD
        return last_visit == today
    except Exception:
        return False


# ─── SQLite Persistence ──────────────────────────────────────────────

def init_db():
    conn = sqlite3.connect(str(DB_PATH))
    conn.execute("""
        CREATE TABLE IF NOT EXISTS visitors (
            username TEXT PRIMARY KEY,
            visit_count INTEGER DEFAULT 0,
            first_seen TEXT,
            last_seen TEXT,
            last_online TEXT,
            last_visited_my_page TEXT,
            last_messaged TEXT,
            message_count INTEGER DEFAULT 0,
            profile_url TEXT,
            last_profile_hash TEXT
        )
    """)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS visit_log (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT,
            session_id TEXT,
            visited_at TEXT,
            status TEXT,
            content_hash TEXT
        )
    """)
    conn.execute("""
        CREATE TABLE IF NOT EXISTS message_log (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT,
            session_id TEXT,
            sent_at TEXT,
            message_text TEXT,
            status TEXT,
            template_index INTEGER
        )
    """)
    conn.commit()
    conn.close()
    log(f"DB initialized: {DB_PATH}")


def record_visitor_seen(username: str, profile_url: str):
    conn = sqlite3.connect(str(DB_PATH))
    now = now_iso()
    existing = conn.execute("SELECT visit_count FROM visitors WHERE username=?", (username,)).fetchone()
    if existing:
        conn.execute(
            "UPDATE visitors SET visit_count=visit_count+1, last_seen=?, profile_url=? WHERE username=?",
            (now, profile_url, username)
        )
    else:
        conn.execute(
            "INSERT INTO visitors (username, visit_count, first_seen, last_seen, profile_url) VALUES (?, 1, ?, ?, ?)",
            (username, now, now, profile_url)
        )
    conn.execute(
        "INSERT INTO visit_log (username, session_id, visited_at, status, content_hash) VALUES (?, ?, ?, 'seen', NULL)",
        (username, SESSION_ID, now)
    )
    conn.commit()
    conn.close()


def get_visitor_stats(username: str) -> dict:
    conn = sqlite3.connect(str(DB_PATH))
    row = conn.execute(
        "SELECT visit_count, first_seen, last_seen, last_online, last_visited_my_page, last_messaged, message_count FROM visitors WHERE username=?",
        (username,)
    ).fetchone()
    conn.close()
    if not row:
        return {"visit_count": 0}
    return {
        "visit_count": row[0],
        "first_seen": row[1],
        "last_seen": row[2],
        "last_online": row[3],
        "last_visited_my_page": row[4],
        "last_messaged": row[5],
        "message_count": row[6],
    }
