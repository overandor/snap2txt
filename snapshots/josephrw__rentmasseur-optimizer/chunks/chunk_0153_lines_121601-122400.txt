def check_auth(authorization: str = None):
    """Auth token validation — fail-closed in production."""
    if not AUTH_TOKEN:
        raise HTTPException(403, {
            "status": "not_configured",
            "message": "Server not configured for uploads. Set JORKI_AUTH_TOKEN secret."
        })
    if not authorization:
        raise HTTPException(401, "Missing Authorization header. Use: Bearer <token>")
    if not authorization.startswith("Bearer "):
        raise HTTPException(401, "Use Bearer token auth")
    token = authorization[7:]
    if token != AUTH_TOKEN:
        raise HTTPException(403, "Invalid auth token")


# --- Google OAuth token verification ---
def verify_google_token(token: str) -> dict:
    """Verify Google ID token using Google's public keys (no credentials.json)."""
    import urllib.request
    # Fetch Google's public certs
    url = "https://www.googleapis.com/oauth2/v3/tokeninfo?id_token=" + token
    try:
        req = urllib.request.Request(url)
        with urllib.request.urlopen(req, timeout=5) as resp:
            data = json.loads(resp.read())
        if GOOGLE_CLIENT_ID and data.get("aud") != GOOGLE_CLIENT_ID:
            raise HTTPException(401, "Token audience mismatch")
        return {"email": data.get("email", ""), "name": data.get("name", ""), "picture": data.get("picture", "")}
    except Exception as e:
        raise HTTPException(401, f"Invalid Google token: {e}")


@app.post("/auth/google")
async def auth_google(body: dict = None):
    """Verify Google ID token and create a session."""
    params = body or {}
    token = params.get("token", "")
    if not token:
        raise HTTPException(400, "Missing 'token' in request body")
    user = verify_google_token(token)
    session_id = hashlib.sha256((user["email"] + str(time.time())).encode()).hexdigest()[:12]
    conn = sqlite3.connect(str(REGISTRY_DB))
    conn.execute("INSERT OR REPLACE INTO sessions VALUES (?,?,?,?,?)",
                 (session_id, "", user["email"], time.time(), "active"))
    conn.commit()
    conn.close()
    return {"session_id": session_id, "user": user, "status": "authenticated"}

async function revokeFile(fid) {
  if (!confirm('Revoke AI access to this file?')) return;

@app.post("/upload")
async def upload_index(
    file: UploadFile = File(...),
    authorization: str = Header(None),
):
    """Upload a compact SQLite index. Returns file_id + query URLs."""
    check_auth(authorization)

    content = await file.read()
    file_id = hashlib.sha256(file.filename.encode() + content[:4096] + str(time.time()).encode()).hexdigest()[:12]
    raw_path = STORAGE_DIR / f"{file_id}.idx"

    # Validate as SQLite, read metadata, then compress for storage
    meta = {}
    try:
        raw_path.write_bytes(content)
        conn = sqlite3.connect(str(raw_path))
        conn.row_factory = sqlite3.Row
        for row in conn.execute("SELECT key, value FROM file_meta"):
            meta[row["key"]] = row["value"]
        conn.close()
        # Compress for storage
        with gzip.open(str(raw_path) + ".gz", "wb") as f:
            f.write(content)
        raw_path.unlink(missing_ok=True)
    except Exception as e:
        raw_path.unlink(missing_ok=True)
        return JSONResponse({"error": f"Invalid index: {e}"}, 400)

    stored_path = str(STORAGE_DIR / f"{file_id}.idx.gz")
    is_compressed = True

    meta["_compressed"] = str(is_compressed)
    save_file_entry(file_id, meta, stored_path)

    base = f"/meta/{file_id}"
    return {
        "file_id": file_id,
        "filename": meta.get("filename", "unknown"),
        "size_human": meta.get("size_human", "?"),
        "merkle_root": meta.get("merkle_root", "?"),
        "format": meta.get("format", "?"),
        "total_lines": meta.get("total_lines", "?"),
        "engine_version": meta.get("engine_version", "1.0"),
        "compressed": is_compressed,
        "query_url": base,
        "sql_url": f"/query/sql/{file_id}",
        "search_url": f"/search/{file_id}",
        "chunk_url": f"/chunk/{file_id}/{{idx}}",
        "mcp_url": f"/mcp",
    }


class IndexConnection:
    """Wraps sqlite3.Connection to clean up temp decompressed files on close."""
    def __init__(self, conn, tmp_path=None):
        self._conn = conn
        self._tmp = tmp_path
    def __getattr__(self, name):
        return getattr(self._conn, name)
    def close(self):
        self._conn.close()
        if self._tmp:
            self._tmp.unlink(missing_ok=True)

def open_index(file_id: str):
    """Open index SQLite DB, handling compression. Cleans up temp files."""
    path = get_file_path(file_id)
    if path.endswith(".gz"):
        decompressed = gzip.decompress(Path(path).read_bytes())
        tmp = STORAGE_DIR / f"_tmp_{file_id}.idx"
        tmp.write_bytes(decompressed)
        conn = sqlite3.connect(str(tmp))
        conn.row_factory = sqlite3.Row
        return IndexConnection(conn, tmp)
    else:
        conn = sqlite3.connect(path)
        conn.row_factory = sqlite3.Row
        return IndexConnection(conn)

def close_index(conn):
    """Close connection and clean up temp decompressed file if present."""
    if isinstance(conn, IndexConnection):
        conn.close()
    else:
        conn.close()

        meta = {row["key"]: row["value"] for row in conn.execute("SELECT key, value FROM file_meta")}

            "SELECT word, count FROM word_freq ORDER BY count DESC LIMIT 20")]

                      for r in conn.execute("SELECT idx, boundary_type, line_start, line_end, size FROM semantic_chunks LIMIT 20")]

        # v2: function index

            funcs = [{"line": r["line_num"], "symbol": r["symbol"]} for r in conn.execute(

            funcs = []

        close_index(conn)

        "filename": meta.get("filename"),
        "size": meta.get("size_human"),
        "format": meta.get("format"),

@app.post("/query/sql/{file_id}")
async def query_sql(file_id: str, body: dict = None):
    conn = open_index(file_id)
    try:
        sql = (body or {}).get("sql", "")
        if not sql:
            raise HTTPException(400, "Missing 'sql' in request body")
        sql_stripped = sql.strip().upper()
        if not sql_stripped.startswith("SELECT") and not sql_stripped.startswith("WITH"):
            raise HTTPException(403, "Only SELECT/WITH queries allowed")
        if any(kw in sql_stripped for kw in ["INSERT", "UPDATE", "DELETE", "DROP", "ALTER", "ATTACH", "PRAGMA"]):
            raise HTTPException(403, "Only read-only queries allowed")
        log_query(file_id, "sql", sql)
        cur = conn.execute(sql)
        columns = [desc[0] for desc in cur.description]
        rows = [dict(zip(columns, [r[c] for c in columns])) for r in cur.fetchall()]
        return {"columns": columns, "rows": rows, "row_count": len(rows)}
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(400, f"SQL error: {e}")
    finally:
        close_index(conn)


@app.post("/query/nosql/{file_id}")
async def query_nosql(file_id: str, body: dict = None):
    conn = open_index(file_id)
    try:
        params = body or {}
        table = params.get("table", "file_meta")
        filter_key = params.get("key")
        filter_value = params.get("value")
        limit = min(params.get("limit", 100), 1000)
        allowed = {
            "file_meta", "chunks", "line_index", "word_freq", "char_freq",
            "line_length_dist", "extracted_entities", "section_headers",
            "code_symbols", "capabilities", "index_stats",
            "semantic_chunks", "bloom_filter", "content_graph", "function_index",
        }
        if table not in allowed:
            raise HTTPException(400, f"Table must be one of: {allowed}")
        # Column allowlist per table
        allowed_columns = {
            "file_meta": {"key", "value"},
            "chunks": {"idx", "offset", "size", "hash", "preview"},
            "line_index": {"line", "offset", "length", "preview"},
            "word_freq": {"word", "count"},
            "char_freq": {"char", "count"},
            "line_length_dist": {"length", "count"},
            "extracted_entities": {"type", "value", "line"},
            "section_headers": {"line", "header"},
            "code_symbols": {"line", "type", "symbol"},
            "capabilities": {"id", "name", "enabled"},
            "index_stats": {"key", "value"},
            "semantic_chunks": {"idx", "offset", "size", "hash", "preview", "boundary_type", "line_start", "line_end"},
            "bloom_filter": {"key", "value"},
            "content_graph": {"source", "target", "relation"},
            "function_index": {"line", "symbol", "args"},
        }
        if filter_key and filter_value:
            if filter_key not in allowed_columns.get(table, set()):
                raise HTTPException(400, f"Column '{filter_key}' not allowed for table '{table}'")
            cur = conn.execute(f"SELECT * FROM {table} WHERE {filter_key} = ? LIMIT ?", [filter_value, limit])
        else:
            cur = conn.execute(f"SELECT * FROM {table} LIMIT ?", [limit])
        columns = [desc[0] for desc in cur.description]
        docs = [dict(zip(columns, [r[c] for c in columns])) for r in cur.fetchall()]
        log_query(file_id, "nosql", f"table={table}")
        return {"table": table, "documents": docs, "count": len(docs)}
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(400, f"Query error: {e}")
    finally:
        close_index(conn)


@app.get("/search/{file_id}")
async def search(file_id: str, q: str = "", limit: int = 50):
    if not q:
        raise HTTPException(400, "Missing query parameter 'q'")
    conn = open_index(file_id)
    try:
        results = {}
        searches = {
            "lines": ("line_index", "preview"),
            "words": ("word_freq", "word"),
            "entities": ("extracted_entities", "value"),
            "symbols": ("code_symbols", "symbol"),
            "headers": ("section_headers", "header"),
            "functions": ("function_index", "symbol"),
            "chunks": ("semantic_chunks", "preview"),
        }
        for key, (table, col) in searches.items():
            try:
                rows = conn.execute(
                    f"SELECT * FROM {table} WHERE {col} LIKE ? LIMIT ?",
                    [f"%{q}%", limit]
                ).fetchall()
                results[key] = [dict(r) for r in rows]
            except:
                results[key] = []
        log_query(file_id, "search", q)
        total = sum(len(v) for v in results.values())
        return {"query": q, "total_matches": total, "results": results}
    finally:
        close_index(conn)
