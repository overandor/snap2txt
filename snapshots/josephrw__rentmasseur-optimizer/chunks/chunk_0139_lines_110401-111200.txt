/*
 * Vercel Serverless Function: /api/collect
 * Collects metrics from the RentMasseur Chrome extension or any client
 * and forwards them to the Hugging Face Space OS backend.
 */

export default async function handler(req, res) {
  if (req.method !== 'POST') {
    return res.status(405).json({ error: 'Method not allowed' });
  }

  const HF_SPACE_URL = process.env.HF_SPACE_URL || '';
  const HF_TOKEN = process.env.HF_TOKEN || '';

  try {
    const payload = req.body;
    const enriched = {
      ...payload,
      source: 'vercel',
      vercel_region: process.env.VERCEL_REGION || 'unknown',
      collected_at: new Date().toISOString(),
    };

    // Forward to HF Space
    if (HF_SPACE_URL) {
      await fetch(`${HF_SPACE_URL}/api/os/ingest`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          ...(HF_TOKEN ? { Authorization: `Bearer ${HF_TOKEN}` } : {}),
        },
        body: JSON.stringify(enriched),
      });
    }

    // Also store in Vercel KV or just return
    return res.status(200).json({
      status: 'collected',
      payload: enriched,
      hf_forwarded: Boolean(HF_SPACE_URL),
    });
  } catch (error) {
    return res.status(500).json({ error: error.message });
  }
}

report.js:
```
/*
 * Vercel Serverless Function: /api/report
 * Returns the full RentMasseur OS report from the HF Space backend.
 */

export default async function handler(req, res) {
  const HF_SPACE_URL = process.env.HF_SPACE_URL || '';
  try {
    if (!HF_SPACE_URL) {
      return res.status(200).json({ note: 'HF_SPACE_URL not set' });
    }
    const resp = await fetch(`${HF_SPACE_URL}/api/os/report`);
    const data = await resp.json();
    return res.status(200).json(data);
  } catch (error) {
    return res.status(500).json({ error: error.message });
  }
}

deploy-hf-space.yml:
```
name: Deploy to Hugging Face Space

on:
  push:
    branches:
      - main
  schedule:
    - cron: '0 */12 * * *'  # Redeploy every 12 hours with new data
  workflow_dispatch:

jobs:
  deploy:
    runs-on: ubuntu-latest
    env:
      HF_TOKEN: ${{ secrets.HF_TOKEN }}
      HF_SPACE_NAME: ${{ secrets.HF_SPACE_NAME }}
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Configure HF credentials
        run: |
          git config --global user.email "github-actions[bot]@users.noreply.github.com"
          git config --global user.name "github-actions[bot]"

      - name: Clone existing HF space
        run: |
          rm -rf hf-space
          git clone https://hf_user:${HF_TOKEN}@huggingface.co/spaces/${HF_SPACE_NAME} hf-space || true

      - name: Sync files to HF space (no secrets in repo)
        run: |
          rm -rf hf-space/*
          rsync -av --exclude='hf-space' --exclude='.git' --exclude='.env' --exclude='debug' --exclude='.github' . hf-space/
          rm -rf hf-space/.git
          rm -rf hf-space/.env
          rm -rf hf-space/debug
          rm -rf hf-space/.github
          # No .env file pushed. Secrets are set via HF Spaces API.

      - name: Set HF Space secrets via API (no secrets in git)
        run: |
          for KEY_VAL in \
            "RENTMASSEUR_USERNAME=${{ secrets.RENTMASSEUR_USERNAME }}" \
            "RENTMASSEUR_PASSWORD=${{ secrets.RENTMASSEUR_PASSWORD }}" \
            "GROQ_API_KEY=${{ secrets.GROQ_API_KEY }}" \
            "GROQ_MODEL=llama-3.3-70b-versatile" \
            "HF_SPACE_NAME=${HF_SPACE_NAME}" \
            "GH_TOKEN=${{ secrets.GH_PAT }}" \
            "GH_REPO=overandor/rentmasseur-extension"; do
            KEY="${KEY_VAL%%=*}"
            VAL="${KEY_VAL#*=}"
            echo "Setting HF Space secret: $KEY"
            curl -s -X POST \
              "https://huggingface.co/api/spaces/${HF_SPACE_NAME}/secrets" \
              -H "Authorization: Bearer ${HF_TOKEN}" \
              -H "Content-Type: application/json" \
              -d "{\"key\":\"$KEY\",\"value\":\"$VAL\"}" || true
          done

      - name: Push to HF space (no .env, no .github, no secrets)
        run: |
          cd hf-space
          rm -f .gitignore
          rm -rf .git
          git init
          git checkout -b main
          git add -A
          git commit -m "Auto-deploy from GitHub Actions $(date -u +%Y-%m-%d-%H:%M:%S)"
          git push https://hf_user:${HF_TOKEN}@huggingface.co/spaces/${HF_SPACE_NAME} main --force

      - name: Write deploy receipt
        run: |
          mkdir -p receipts
          cat > receipts/deploy_$(date -u +%Y%m%d_%H%M%S).json <<EOF
          {
            "action": "deploy-hf-space",
            "timestamp": "$(date -u +%Y-%m-%dT%H:%M:%SZ)",
            "commit": "${{ github.sha }}",
            "space": "${HF_SPACE_NAME}",
            "secrets_pushed_to_git": false,
            "secrets_set_via_api": true,
            "status": "pending_verification"
          }
          EOF
          git add receipts/ || true
          git -c user.name="github-actions[bot]" -c user.email="github-actions[bot]@users.noreply.github.com" commit -m "Deploy receipt $(date -u +%Y-%m-%d-%H:%M)" || true
          git push || true

      - name: Verify HF Space health (hard pass/fail with retries)
        run: |
          # Derive public HF Space URL from HF_SPACE_NAME (owner/space -> owner-space.hf.space)
          SPACE_URL=$(echo "${HF_SPACE_NAME}" | sed 's|/|-|g')
          HEALTH_URL="https://${SPACE_URL}.hf.space/api/health"
          echo "HF Space: ${HF_SPACE_NAME}"
          echo "Health URL: ${HEALTH_URL}"

          # Wait for HF Docker build + startup (can take 60-180s)
          sleep 90

          for i in 1 2 3 4 5; do
            echo "Health check attempt $i..."
            HEALTH=$(curl -s -o /tmp/health_response -w "%{http_code}" --max-time 30 "${HEALTH_URL}")
            echo "HF Space health endpoint: $HEALTH"
            if [ "$HEALTH" == "200" ]; then
              echo "PASS: HF Space is live and healthy"
              cat /tmp/health_response
              exit 0
            fi
            echo "Response body:"
            cat /tmp/health_response || true
            echo "Waiting 60s before retry..."
            sleep 60
          done

          echo "FAIL: HF Space health endpoint did not return 200 after 5 attempts"
          exit 1

        run: |
          set -euo pipefail
