@app.post("/upload")
async def upload_index(
    file: UploadFile = File(...),
    authorization: str = Header(None),
):
    """Upload a compact SQLite index. Returns file_id + query URLs."""
    check_auth(authorization)

    content = await file.read()
    file_id = hashlib.sha256(file.filename.encode() + content[:4096] + str(time.time()).encode()).hexdigest()[:12]
    raw_path = STORAGE_DIR / f"{file_id}.idx"

    # Validate as SQLite, read metadata, then compress for storage
    meta = {}
    try:
        raw_path.write_bytes(content)
        conn = sqlite3.connect(str(raw_path))
        conn.row_factory = sqlite3.Row
        for row in conn.execute("SELECT key, value FROM file_meta"):
            meta[row["key"]] = row["value"]
        conn.close()
        # Compress for storage
        with gzip.open(str(raw_path) + ".gz", "wb") as f:
            f.write(content)
        raw_path.unlink(missing_ok=True)
    except Exception as e:
        raw_path.unlink(missing_ok=True)
        return JSONResponse({"error": f"Invalid index: {e}"}, 400)

    stored_path = str(STORAGE_DIR / f"{file_id}.idx.gz")
    is_compressed = True

    meta["_compressed"] = str(is_compressed)
    save_file_entry(file_id, meta, stored_path)

    base = f"/meta/{file_id}"
    return {
        "file_id": file_id,
        "filename": meta.get("filename", "unknown"),
        "size_human": meta.get("size_human", "?"),
        "merkle_root": meta.get("merkle_root", "?"),
        "format": meta.get("format", "?"),
        "total_lines": meta.get("total_lines", "?"),
        "engine_version": meta.get("engine_version", "1.0"),
        "compressed": is_compressed,
        "query_url": base,
        "sql_url": f"/query/sql/{file_id}",
        "search_url": f"/search/{file_id}",
        "chunk_url": f"/chunk/{file_id}/{{idx}}",
        "mcp_url": f"/mcp",
    }


class IndexConnection:
    """Wraps sqlite3.Connection to clean up temp decompressed files on close."""
    def __init__(self, conn, tmp_path=None):
        self._conn = conn
        self._tmp = tmp_path
    def __getattr__(self, name):
        return getattr(self._conn, name)
    def close(self):
        self._conn.close()
        if self._tmp:
            self._tmp.unlink(missing_ok=True)

def open_index(file_id: str):
    """Open index SQLite DB, handling compression. Cleans up temp files."""
    path = get_file_path(file_id)
    if path.endswith(".gz"):
        decompressed = gzip.decompress(Path(path).read_bytes())
        tmp = STORAGE_DIR / f"_tmp_{file_id}.idx"
        tmp.write_bytes(decompressed)
        conn = sqlite3.connect(str(tmp))
        conn.row_factory = sqlite3.Row
        return IndexConnection(conn, tmp)
    else:
        conn = sqlite3.connect(path)
        conn.row_factory = sqlite3.Row
        return IndexConnection(conn)

def close_index(conn):
    """Close connection and clean up temp decompressed file if present."""
    if isinstance(conn, IndexConnection):
        conn.close()
    else:
        conn.close()

@app.get("/summary/{file_id}")
async def get_summary(file_id: str):
    conn = open_index(file_id)
    try:
        meta = {row["key"]: row["value"] for row in conn.execute("SELECT key, value FROM file_meta")}
        head = meta.get("head_sample_4kb", "")
        tail = meta.get("tail_sample_4kb", "")
        words = [{"word": r["word"], "count": r["count"]} for r in conn.execute(
            "SELECT word, count FROM word_freq ORDER BY count DESC LIMIT 20")]
        lengths = [{"bucket": r["bucket"], "count": r["count"]} for r in conn.execute(
            "SELECT bucket, count FROM line_length_dist")]
        headers = [{"header": r["header"], "count": r["count"]} for r in conn.execute(
            "SELECT header, count FROM section_headers ORDER BY count DESC LIMIT 20")]
        # v2: semantic chunk summary
        try:
            chunks = [{"idx": r["idx"], "type": r["boundary_type"],
                       "lines": f"{r['line_start']}-{r['line_end']}", "size": r["size"]}
                      for r in conn.execute("SELECT idx, boundary_type, line_start, line_end, size FROM semantic_chunks LIMIT 20")]
        except:
            chunks = []
        # v2: function index
        try:
            funcs = [{"line": r["line_num"], "symbol": r["symbol"]} for r in conn.execute(
                "SELECT line_num, symbol FROM function_index LIMIT 20")]
        except:
            funcs = []
    finally:
        close_index(conn)

    return {
        "filename": meta.get("filename"),
        "size": meta.get("size_human"),
        "format": meta.get("format"),
        "total_lines": meta.get("total_lines"),
        "entropy": meta.get("entropy"),
        "merkle_root": meta.get("merkle_root"),
        "head_4kb": head[:2000],
        "tail_4kb": tail[:2000],
        "top_words": words,
        "line_length_distribution": lengths,
        "section_headers": headers,
        "semantic_chunks": chunks,
        "functions": funcs,
    }

@app.get("/chunks/{file_id}")
async def list_chunks(file_id: str, boundary_type: str = None, limit: int = 100):

    conn = open_index(file_id)
    try:
        if boundary_type:
            rows = conn.execute(
                "SELECT idx, offset, size, boundary_type, line_start, line_end, substr(preview,1,80) as preview_short FROM semantic_chunks WHERE boundary_type = ? ORDER BY idx LIMIT ?",
                [boundary_type, min(limit, 1000)]
            ).fetchall()
        else:
            rows = conn.execute(
                "SELECT idx, offset, size, boundary_type, line_start, line_end, substr(preview,1,80) as preview_short FROM semantic_chunks ORDER BY idx LIMIT ?",
                [min(limit, 1000)]
            ).fetchall()
        return {
            "file_id": file_id,
            "total": len(rows),
            "chunks": [dict(r) for r in rows],
        }
    except sqlite3.OperationalError:
        return {"file_id": file_id, "total": 0, "chunks": [], "note": "No semantic_chunks table (v1 index)"}
    finally:
        close_index(conn)

@app.post("/query/sql/{file_id}")
async def query_sql(file_id: str, body: dict = None):
    conn = open_index(file_id)
    try:
        sql = (body or {}).get("sql", "")
        if not sql:
            raise HTTPException(400, "Missing 'sql' in request body")
        sql_stripped = sql.strip().upper()
        if not sql_stripped.startswith("SELECT") and not sql_stripped.startswith("WITH"):
            raise HTTPException(403, "Only SELECT/WITH queries allowed")
        if any(kw in sql_stripped for kw in ["INSERT", "UPDATE", "DELETE", "DROP", "ALTER", "ATTACH", "PRAGMA"]):
            raise HTTPException(403, "Only read-only queries allowed")
        log_query(file_id, "sql", sql)
        cur = conn.execute(sql)
        columns = [desc[0] for desc in cur.description]
        rows = [dict(zip(columns, [r[c] for c in columns])) for r in cur.fetchall()]
        return {"columns": columns, "rows": rows, "row_count": len(rows)}
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(400, f"SQL error: {e}")
    finally:
        close_index(conn)


@app.post("/query/nosql/{file_id}")
async def query_nosql(file_id: str, body: dict = None):
    conn = open_index(file_id)
    try:
        params = body or {}
        table = params.get("table", "file_meta")
        filter_key = params.get("key")
        filter_value = params.get("value")
        limit = min(params.get("limit", 100), 1000)
        allowed = {
            "file_meta", "chunks", "line_index", "word_freq", "char_freq",
            "line_length_dist", "extracted_entities", "section_headers",
            "code_symbols", "capabilities", "index_stats",
            "semantic_chunks", "bloom_filter", "content_graph", "function_index",
        }
        if table not in allowed:
            raise HTTPException(400, f"Table must be one of: {allowed}")
        # Column allowlist per table
        allowed_columns = {
            "file_meta": {"key", "value"},
            "chunks": {"idx", "offset", "size", "hash", "preview"},
            "line_index": {"line", "offset", "length", "preview"},
            "word_freq": {"word", "count"},
            "char_freq": {"char", "count"},
            "line_length_dist": {"length", "count"},
            "extracted_entities": {"type", "value", "line"},
            "section_headers": {"line", "header"},
            "code_symbols": {"line", "type", "symbol"},
            "capabilities": {"id", "name", "enabled"},
            "index_stats": {"key", "value"},
            "semantic_chunks": {"idx", "offset", "size", "hash", "preview", "boundary_type", "line_start", "line_end"},
            "bloom_filter": {"key", "value"},
            "content_graph": {"source", "target", "relation"},
            "function_index": {"line", "symbol", "args"},
        }
        if filter_key and filter_value:
            if filter_key not in allowed_columns.get(table, set()):
                raise HTTPException(400, f"Column '{filter_key}' not allowed for table '{table}'")
            cur = conn.execute(f"SELECT * FROM {table} WHERE {filter_key} = ? LIMIT ?", [filter_value, limit])
        else:
            cur = conn.execute(f"SELECT * FROM {table} LIMIT ?", [limit])
        columns = [desc[0] for desc in cur.description]
        docs = [dict(zip(columns, [r[c] for c in columns])) for r in cur.fetchall()]
        log_query(file_id, "nosql", f"table={table}")
        return {"table": table, "documents": docs, "count": len(docs)}
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(400, f"Query error: {e}")
    finally:
        close_index(conn)


@app.get("/search/{file_id}")
async def search(file_id: str, q: str = "", limit: int = 50):
    if not q:
        raise HTTPException(400, "Missing query parameter 'q'")
    conn = open_index(file_id)
    try:
        results = {}
        searches = {
            "lines": ("line_index", "preview"),
            "words": ("word_freq", "word"),
            "entities": ("extracted_entities", "value"),
            "symbols": ("code_symbols", "symbol"),
            "headers": ("section_headers", "header"),
            "functions": ("function_index", "symbol"),
            "chunks": ("semantic_chunks", "preview"),
        }
        for key, (table, col) in searches.items():
            try:
                rows = conn.execute(
                    f"SELECT * FROM {table} WHERE {col} LIKE ? LIMIT ?",
                    [f"%{q}%", limit]
                ).fetchall()
                results[key] = [dict(r) for r in rows]
            except:
                results[key] = []
        log_query(file_id, "search", q)
        total = sum(len(v) for v in results.values())
        return {"query": q, "total_matches": total, "results": results}
    finally:
        close_index(conn)

@app.post("/mcp/query")
async def mcp_query(body: dict = None):
    """MCP query endpoint — unified query interface for AI tools."""
    params = body or {}
    action = params.get("action", "search")
    file_id = params.get("file_id", "")
    query = params.get("query", "")

    if action == "list_files" or not file_id:
        reg = load_registry()
        files = [{"file_id": fid, "filename": info["filename"]} for fid, info in reg.items()]
        if not file_id and action != "list_files":
            if not files:
                return {"files": [], "message": "No files indexed yet. Upload a .idx file first."}
            # Auto-select first file for console convenience
            file_id = files[0]["file_id"]
        else:
            return {"files": files}

    if action == "search":
        return await search(file_id, q=query)
    elif action == "summary":
        return await get_summary(file_id)
    elif action == "meta":
        return await get_meta(file_id)
    elif action == "chunk":
        idx = params.get("chunk_idx", 0)
        return await get_chunk(file_id, idx)
    elif action == "sql":
        return await query_sql(file_id, {"sql": query})
    elif action == "capabilities":
        return await get_capabilities(file_id)
    elif action == "list_chunks":
        return await list_chunks(file_id)
    else:
        raise HTTPException(400, f"Unknown action: {action}. Use: search, summary, meta, chunk, sql, capabilities, list_chunks, list_files")

async def get_stats(file_id: str):
    """Query statistics for a file."""
    conn = sqlite3.connect(str(REGISTRY_DB))
    conn.row_factory = sqlite3.Row
    rows = conn.execute(
        "SELECT query_type, COUNT(*) as count, MAX(timestamp) as last_query FROM query_log WHERE file_id = ? GROUP BY query_type",

    ).fetchall()

    return {"file_id": file_id, "stats": [dict(r) for r in rows]}


@app.delete("/revoke/{file_id}")
async def revoke_access(file_id: str, authorization: str = Header(None)):
    """Revoke AI access to a file by deleting its index."""
    check_auth(authorization)
    reg = load_registry()
    if file_id not in reg:
        raise HTTPException(404, {
            "status": "session_not_found",
            "file_id": file_id,
            "message": "This Jorki session is expired, revoked, or unavailable.",
            "next": "Create a new Jorki session."
        })
    path = reg[file_id]["path"]
    try:
        Path(path).unlink(missing_ok=True)
    except:
        pass
    conn = sqlite3.connect(str(REGISTRY_DB))
    conn.execute("DELETE FROM files WHERE file_id = ?", [file_id])
    conn.execute("DELETE FROM query_log WHERE file_id = ?", [file_id])
    conn.commit()
    conn.close()
    return {"status": "revoked", "file_id": file_id}

@app.post("/graphql")
async def graphql(body: dict = None):
    """Minimal GraphQL-like query endpoint."""
    params = body or {}
    query = params.get("query", "")
    # Parse simple GraphQL: { file(id: "xxx") { filename size format } }
    import re
    file_match = re.search(r'file\s*\(\s*id:\s*"([^"]+)"\s*\)', query)
    if file_match:
        fid = file_match.group(1)
        reg = load_registry()
        if fid not in reg:
            return {"errors": [{"message": f"file_id '{fid}' not found"}]}
        info = reg[fid]
        meta = info["meta"]
        fields = re.findall(r'(\w+)', query.split("}", 1)[0])
        result = {}
        for f in fields:
            if f in ("filename",): result["filename"] = meta.get("filename", "unknown")
            elif f in ("size", "size_human"): result["size"] = meta.get("size_human", "?")
            elif f in ("format",): result["format"] = meta.get("format", "?")
            elif f in ("merkle", "merkle_root"): result["merkle_root"] = meta.get("merkle_root", "?")
            elif f in ("lines", "total_lines"): result["total_lines"] = meta.get("total_lines", "?")
            elif f in ("entropy",): result["entropy"] = meta.get("entropy", "?")
            elif f in ("engine", "engine_version"): result["engine_version"] = meta.get("engine_version", "1.0")
        return {"data": {"file": result}}
    # List all files
    if "files" in query:
        reg = load_registry()
        return {"data": {"files": [{"file_id": fid, "filename": info["filename"]} for fid, info in reg.items()]}}
    return {"errors": [{"message": "Unsupported query. Use: { file(id: \"xxx\") { filename size format } } or { files { filename } }"}]}


# --- Clipboard Superposition ---
# Multiple Jorki sessions encoded into a single base64 blob for clipboard.
# Format: JORKI:v1:<base64(json({sessions: [{file_id, filename, url, added_at}]}))>

SUPERPOSE_PREFIX = "JORKI:v1:"

def encode_superposition(file_ids: list) -> str:
    """Pack multiple file_ids into a single base64 clipboard string."""
    reg = load_registry()
    sessions = []
    for fid in file_ids:
        if fid in reg:
            info = reg[fid]
            sessions.append({
                "file_id": fid,
                "filename": info["meta"].get("filename", "unknown"),
                "size": info["meta"].get("size_human", "?"),
                "url": f"/meta/{fid}",
                "added_at": time.time(),
            })
    payload = json.dumps({"sessions": sessions, "created_at": time.time()}).encode()
    return SUPERPOSE_PREFIX + base64.b64encode(payload).decode()

def decode_superposition(blob: str) -> dict:
    """Unpack a superposition blob into session list."""
    blob = blob.strip()
    if not blob.startswith(SUPERPOSE_PREFIX):
        raise HTTPException(400, {"status": "invalid_blob", "message": "Not a Jorki superposition blob."})
    try:
        payload = base64.b64decode(blob[len(SUPERPOSE_PREFIX):])
        return json.loads(payload)
    except Exception:
        raise HTTPException(400, {"status": "invalid_blob", "message": "Corrupted superposition blob."})

@app.post("/superpose/encode")
async def superpose_encode(file_ids: list = Body(...)):
    """Encode multiple file_ids into a single clipboard-ready base64 blob.
    Paste the blob into any LLM — it can decode and query all files at once."""
    if not file_ids:
        raise HTTPException(400, {"status": "error", "message": "Provide a list of file_ids."})
    blob = encode_superposition(file_ids)
    reg = load_registry()
    valid = [fid for fid in file_ids if fid in reg]
    return {
        "blob": blob,
        "sessions": len(valid),
        "file_ids": valid,
        "format": "JORKI:v1:base64",
        "usage": "Copy the blob field. Paste into any LLM. The LLM can POST it to /superpose/query to query all files at once.",
    }

@app.post("/superpose/decode")
async def superpose_decode(blob: str = Body(..., embed=True)):
    """Decode a superposition blob back into session list with live status."""
    data = decode_superposition(blob)
    reg = load_registry()
    for s in data["sessions"]:
        s["live"] = s["file_id"] in reg
        s["status"] = "live" if s["live"] else "expired"
    data["active_sessions"] = sum(1 for s in data["sessions"] if s["live"])
    data["expired_sessions"] = sum(1 for s in data["sessions"] if not s["live"])
    return data

@app.post("/superpose/query")
async def superpose_query(body: dict = Body(...)):
    """Query across ALL files in a superposition blob simultaneously.
    The LLM posts the blob + a query, gets aggregated results from every file."""
    blob = body.get("blob", "")
    query = body.get("query", "")
    action = body.get("action", "search")

    if not blob:
        raise HTTPException(400, {"status": "error", "message": "Provide a superposition blob."})

    data = decode_superposition(blob)
    reg = load_registry()
    results = []
    total_matches = 0

    for s in data["sessions"]:
        fid = s["file_id"]
        if fid not in reg:
            results.append({
                "file_id": fid,
                "filename": s.get("filename", "?"),
                "status": "expired",
                "matches": 0,
            })
            continue

        try:
            if action == "search" and query:
                conn = open_index(fid)
                try:
                    file_results = {}
                    searches = {
                        "lines": ("line_index", "preview"),
                        "words": ("word_freq", "word"),
                        "symbols": ("code_symbols", "symbol"),
                        "headers": ("section_headers", "header"),
                        "functions": ("function_index", "symbol"),
                        "chunks": ("semantic_chunks", "preview"),
                    }
                    for key, (table, col) in searches.items():
                        try:
                            rows = conn.execute(
                                f"SELECT * FROM {table} WHERE {col} LIKE ? LIMIT 10",
                                [f"%{query}%"]
                            ).fetchall()
                            if rows:
                                file_results[key] = [dict(r) for r in rows]
                        except:
                            pass
                    matches = sum(len(v) for v in file_results.values())
                    total_matches += matches
                    results.append({
                        "file_id": fid,
                        "filename": s.get("filename", reg[fid]["meta"].get("filename", "?")),
                        "status": "live",
                        "matches": matches,
                        "results": file_results,
                    })
                finally:
                    close_index(conn)
            elif action == "meta":
                results.append({
                    "file_id": fid,
                    "filename": reg[fid]["meta"].get("filename", "?"),
                    "status": "live",
                    "meta": reg[fid]["meta"],
                })
                total_matches += 1
            elif action == "summary":
                conn = open_index(fid)
                try:
                    meta = {row["key"]: row["value"] for row in conn.execute("SELECT key, value FROM file_meta")}
                    chunks_count = conn.execute("SELECT COUNT(*) FROM semantic_chunks").fetchone()[0]
                    results.append({
                        "file_id": fid,
                        "filename": reg[fid]["meta"].get("filename", "?"),
                        "status": "live",
                        "total_lines": meta.get("total_lines"),
                        "chunks": chunks_count,
                        "format": meta.get("format"),
                    })
                    total_matches += 1
                except:
                    results.append({"file_id": fid, "status": "live", "error": "summary failed"})
                finally:
                    close_index(conn)
            elif action == "state":
                stats_conn = sqlite3.connect(str(REGISTRY_DB))
                stats_conn.row_factory = sqlite3.Row
                stats = stats_conn.execute(
                    "SELECT query_type, COUNT(*) as count FROM query_log WHERE file_id = ? GROUP BY query_type",
                    [fid]
                ).fetchall()
                stats_conn.close()
                results.append({
                    "file_id": fid,
                    "filename": reg[fid]["meta"].get("filename", "?"),
                    "status": "live",
                    "queries": [dict(r) for r in stats],
                    "total_queries": sum(r["count"] for r in stats),
                })
                total_matches += 1
        except Exception as e:
            results.append({
                "file_id": fid,
                "filename": s.get("filename", "?"),
                "status": "error",
                "error": str(e),
            })

    return {
        "blob_sessions": len(data["sessions"]),
        "active": sum(1 for r in results if r.get("status") == "live"),
        "expired": sum(1 for r in results if r.get("status") == "expired"),
        "total_matches": total_matches,
        "query": query,
        "action": action,
        "results": results,
    }
