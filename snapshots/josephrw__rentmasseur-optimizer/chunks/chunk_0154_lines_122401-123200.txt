@app.post("/graphql")
async def graphql(body: dict = None):
    """Minimal GraphQL-like query endpoint."""
    params = body or {}
    query = params.get("query", "")
    # Parse simple GraphQL: { file(id: "xxx") { filename size format } }
    import re
    file_match = re.search(r'file\s*\(\s*id:\s*"([^"]+)"\s*\)', query)
    if file_match:
        fid = file_match.group(1)
        reg = load_registry()
        if fid not in reg:
            return {"errors": [{"message": f"file_id '{fid}' not found"}]}
        info = reg[fid]
        meta = info["meta"]
        fields = re.findall(r'(\w+)', query.split("}", 1)[0])
        result = {}
        for f in fields:
            if f in ("filename",): result["filename"] = meta.get("filename", "unknown")
            elif f in ("size", "size_human"): result["size"] = meta.get("size_human", "?")
            elif f in ("format",): result["format"] = meta.get("format", "?")
            elif f in ("merkle", "merkle_root"): result["merkle_root"] = meta.get("merkle_root", "?")
            elif f in ("lines", "total_lines"): result["total_lines"] = meta.get("total_lines", "?")
            elif f in ("entropy",): result["entropy"] = meta.get("entropy", "?")
            elif f in ("engine", "engine_version"): result["engine_version"] = meta.get("engine_version", "1.0")
        return {"data": {"file": result}}
    # List all files
    if "files" in query:
        reg = load_registry()
        return {"data": {"files": [{"file_id": fid, "filename": info["filename"]} for fid, info in reg.items()]}}
    return {"errors": [{"message": "Unsupported query. Use: { file(id: \"xxx\") { filename size format } } or { files { filename } }"}]}


# --- Clipboard Superposition ---
# Multiple Jorki sessions encoded into a single base64 blob for clipboard.
# Format: JORKI:v1:<base64(json({sessions: [{file_id, filename, url, added_at}]}))>

SUPERPOSE_PREFIX = "JORKI:v1:"

def encode_superposition(file_ids: list) -> str:
    """Pack multiple file_ids into a single base64 clipboard string."""
    reg = load_registry()
    sessions = []
    for fid in file_ids:
        if fid in reg:
            info = reg[fid]
            sessions.append({
                "file_id": fid,
                "filename": info["meta"].get("filename", "unknown"),
                "size": info["meta"].get("size_human", "?"),
                "url": f"/meta/{fid}",
                "added_at": time.time(),
            })
    payload = json.dumps({"sessions": sessions, "created_at": time.time()}).encode()
    return SUPERPOSE_PREFIX + base64.b64encode(payload).decode()

def decode_superposition(blob: str) -> dict:
    """Unpack a superposition blob into session list."""
    blob = blob.strip()
    if not blob.startswith(SUPERPOSE_PREFIX):
        raise HTTPException(400, {"status": "invalid_blob", "message": "Not a Jorki superposition blob."})
    try:
        payload = base64.b64decode(blob[len(SUPERPOSE_PREFIX):])
        return json.loads(payload)
    except Exception:
        raise HTTPException(400, {"status": "invalid_blob", "message": "Corrupted superposition blob."})

@app.post("/superpose/encode")
async def superpose_encode(file_ids: list = Body(...)):
    """Encode multiple file_ids into a single clipboard-ready base64 blob.

        "usage": "Copy the blob field. Paste into any LLM. The LLM can POST it to /superpose/query to query all files at once.",
    }

@app.post("/superpose/decode")
async def superpose_decode(blob: str = Body(..., embed=True)):
    """Decode a superposition blob back into session list with live status."""
    data = decode_superposition(blob)
    reg = load_registry()
    for s in data["sessions"]:
        s["live"] = s["file_id"] in reg
        s["status"] = "live" if s["live"] else "expired"
    data["active_sessions"] = sum(1 for s in data["sessions"] if s["live"])
    data["expired_sessions"] = sum(1 for s in data["sessions"] if not s["live"])
    return data

@app.post("/superpose/query")
async def superpose_query(body: dict = Body(...)):
    """Query across ALL files in a superposition blob simultaneously.
    The LLM posts the blob + a query, gets aggregated results from every file."""
    blob = body.get("blob", "")
    query = body.get("query", "")
    action = body.get("action", "search")

    if not blob:
        raise HTTPException(400, {"status": "error", "message": "Provide a superposition blob."})

    data = decode_superposition(blob)
    reg = load_registry()
    results = []
    total_matches = 0

    for s in data["sessions"]:
        fid = s["file_id"]
        if fid not in reg:
            results.append({
                "file_id": fid,
                "filename": s.get("filename", "?"),
                "status": "expired",
                "matches": 0,
            })
            continue

        try:
            if action == "search" and query:
                conn = open_index(fid)
                try:
                    file_results = {}
                    searches = {
                        "lines": ("line_index", "preview"),
                        "words": ("word_freq", "word"),
                        "symbols": ("code_symbols", "symbol"),
                        "headers": ("section_headers", "header"),
                        "functions": ("function_index", "symbol"),
                        "chunks": ("semantic_chunks", "preview"),
                    }
                    for key, (table, col) in searches.items():
                        try:
                            rows = conn.execute(
                                f"SELECT * FROM {table} WHERE {col} LIKE ? LIMIT 10",
                                [f"%{query}%"]
                            ).fetchall()
                            if rows:
                                file_results[key] = [dict(r) for r in rows]
                        except:
                            pass
                    matches = sum(len(v) for v in file_results.values())
                    total_matches += matches
                    results.append({
                        "file_id": fid,
                        "filename": s.get("filename", reg[fid]["meta"].get("filename", "?")),
                        "status": "live",
                        "matches": matches,
                        "results": file_results,
                    })
                finally:
                    close_index(conn)
            elif action == "meta":
                results.append({
                    "file_id": fid,
                    "filename": reg[fid]["meta"].get("filename", "?"),
                    "status": "live",
                    "meta": reg[fid]["meta"],
                })
                total_matches += 1
            elif action == "summary":
                conn = open_index(fid)
                try:
                    meta = {row["key"]: row["value"] for row in conn.execute("SELECT key, value FROM file_meta")}
                    chunks_count = conn.execute("SELECT COUNT(*) FROM semantic_chunks").fetchone()[0]
                    results.append({
                        "file_id": fid,
                        "filename": reg[fid]["meta"].get("filename", "?"),
                        "status": "live",
                        "total_lines": meta.get("total_lines"),
                        "chunks": chunks_count,
                        "format": meta.get("format"),
                    })
                    total_matches += 1
                except:
                    results.append({"file_id": fid, "status": "live", "error": "summary failed"})
                finally:
                    close_index(conn)
            elif action == "state":
                stats_conn = sqlite3.connect(str(REGISTRY_DB))
                stats_conn.row_factory = sqlite3.Row
                stats = stats_conn.execute(
                    "SELECT query_type, COUNT(*) as count FROM query_log WHERE file_id = ? GROUP BY query_type",
                    [fid]
                ).fetchall()
                stats_conn.close()
                results.append({
                    "file_id": fid,
                    "filename": reg[fid]["meta"].get("filename", "?"),
                    "status": "live",
                    "queries": [dict(r) for r in stats],
                    "total_queries": sum(r["count"] for r in stats),
                })
                total_matches += 1
        except Exception as e:
            results.append({
                "file_id": fid,
                "filename": s.get("filename", "?"),
                "status": "error",
                "error": str(e),
            })

    return {
        "blob_sessions": len(data["sessions"]),
        "active": sum(1 for r in results if r.get("status") == "live"),
        "expired": sum(1 for r in results if r.get("status") == "expired"),
        "total_matches": total_matches,
        "query": query,
        "action": action,
        "results": results,
    }
