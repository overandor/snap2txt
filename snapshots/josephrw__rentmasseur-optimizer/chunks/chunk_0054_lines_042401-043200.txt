                    self.api.put("/settings/about", {"headline": suggestion["headline"], "description": suggestion["bio"]})
                    after = {"headline": suggestion["headline"], "description": suggestion["bio"][:200]}
                    conn = db()
                    conn.execute("UPDATE bio_versions SET applied=1 WHERE ts=(SELECT ts FROM bio_versions ORDER BY ts DESC LIMIT 1)")
                    conn.commit()
                    conn.close()
                    receipt("apply_bio", True, f"Applied LLM bio: {suggestion['headline']}", before, after)
                else:
                    log.info("Bio application rejected by governor")
            else:
                receipt("draft_bio", True, f"Drafted bio: {suggestion['headline']}")
        except Exception as e:
            receipt("bio_experiment", False, str(e))
            log.error("Bio experiment failed: %s", e)

    def health_check(self):
        try:
            self.api.ensure_session()
            resp = self.api.get("/account/dashboard")
            if resp.status_code == 200:
                receipt("health_check", True, "Session healthy")
            else:
                record_alert("error", "health", f"Dashboard returned {resp.status_code}")
        except Exception as e:
            record_alert("error", "health", str(e))

    def cycle(self):
        now = time.time()
        if now - self.last_visibility >= VISIBILITY_CHECK_INTERVAL:
            self.ensure_visible()
            self.last_visibility = now
        if now - self.last_avail >= AVAIL_CHECK_INTERVAL:
            self.ensure_available()
            self.last_avail = now
        if now - self.last_stats >= STATS_INTERVAL:
            self.collect_stats()
            self.last_stats = now
        if now - self.last_bio >= BIO_EXPERIMENT_INTERVAL:
            self.bio_experiment(auto_apply=False)
            self.last_bio = now
        if now - self.last_health >= HEALTH_CHECK_INTERVAL:
            self.health_check()
            self.last_health = now
        if now - self.last_dashboard >= DASHBOARD_INTERVAL:
            self.print_dashboard()
            self.last_dashboard = now

            print(f"  Views: {prof.get('totalPageViews', 'N/A')}")
            print(f"  Contacts: {prof.get('totalContactClicks', 'N/A')}")
            print(f"  Visits: {keep.get('newVisits', 'N/A')}")
            print(f"  Emails: {keep.get('newEmails', 'N/A')}")
            print(f"  Bookmarks: {dash.get('onlineBookmarks', 'N/A')}")
            print(f"  Headline: {assets.get('headline', 'N/A')}")
            print(f"  Bio: {len(assets.get('description', ''))} chars")
            print(f"  Rate interval: {self.api.rate_limiter.current_interval:.1f}s")
            print(f"{'='*60}\n")
        except Exception as e:
            log.error("Dashboard print failed: %s", e)

    def run_once(self):
        if not self.login():
            print("Login failed")
            return
        self.print_dashboard()
        self.ensure_visible()
        self.ensure_available()
        self.collect_stats()
        self.bio_experiment(auto_apply=False)

    def run_daemon(self):
        log.info("=== FORTRESS daemon starting ===")
        if not self.login():
            log.error("Initial login failed")
            return
        self.print_dashboard()
        while self.running:
            try:
                self.cycle()
            except KeyboardInterrupt:
                log.info("Shutdown")
                self.running = False
                break
            except Exception as e:
                log.error("Cycle error: %s", e)
                receipt("daemon_cycle", False, str(e))
                time.sleep(60)
            time.sleep(60)
        log.info("Fortress stopped")

"""

def collect_snapshot(api, tenant_id: str = "") -> MetricSnapshot:
    """Collect a metric snapshot from all verified endpoints."""
    snap = MetricSnapshot(
        tenant_id=tenant_id,
        ts=datetime.now(timezone.utc).isoformat(),
    )

    # Dashboard
    try:
        dash = api.get_dashboard()
        ps = dash.get("profileStatistics") or {}
        snap.views = int(ps.get("totalPageViews", 0))
        snap.contact_clicks = int(ps.get("totalContactClicks", 0))
        snap.is_hidden = bool(dash.get("isAdHidden", False))
    except Exception as e:
        log.warning(f"dashboard error: {e}")

    # Ad statistics
    try:
        stats = api.get_ad_statistics()
        ps = stats.get("profileStatistics") or {}
        if not snap.views:
            snap.views = int(ps.get("totalPageViews", 0))
        if not snap.contact_clicks:
            snap.contact_clicks = int(ps.get("totalContactClicks", 0))
    except Exception as e:
        log.warning(f"ad_statistics error: {e}")

    # KeepOnline
    try:
        keep = api.get_keeponline()
        snap.new_emails = int(keep.get("newEmails", 0))
        snap.new_visits = int(keep.get("newVisits", 0))
        snap.is_hidden = snap.is_hidden or bool(keep.get("isAdHidden", False))
    except Exception as e:
        log.warning(f"keeponline error: {e}")

    # Availability
    try:
        avail = api.get_availability()
        selected = avail.get("selected", "")
        snap.is_available = (selected == "Available")
        countdown = int(avail.get("countdown", 0) or 0)
        if countdown:
            snap.available_minutes = max(0, (countdown - time.time()) / 60.0)
    except Exception as e:
        log.warning(f"availability error: {e}")

    # About
    try:
        about = api.get_about()
        props = about.get("userProps", {})
        assets = props.get("assets", {})
        headline = assets.get("headline", "")
        info = props.get("information", {})
        bio_text = info.get("description", "") or ""
        snap.headline_hash = hashlib.sha256(headline.encode()).hexdigest()[:8] if headline else ""
        snap.bio_hash = hashlib.sha256(bio_text.encode()).hexdigest()[:8] if bio_text else ""
    except Exception as e:
        log.warning(f"about error: {e}")

    # Mailbox
    try:
        mailbox = api.get_mailbox(page=1, folder=1, sort=1)
        emails = mailbox.get("emails", [])
        snap.emails_count = len(emails)

        booking_kw = ["book", "today", "tonight", "tomorrow", "session", "schedule", "available"]
        repeat_kw = ["again", "last time", "saw you", "been a while", "regular", "missed you"]

        for email in emails:
            msg = (email.get("lastMessage", "") or "").lower()
            if any(kw in msg for kw in booking_kw):
                snap.booking_proxy_count += 1
            if any(kw in msg for kw in repeat_kw):
                snap.repeat_sender_count += 1

        if emails:
            scores = []
            for email in emails:
                msg = (email.get("lastMessage", "") or "").lower()
                score = 0.0
                if any(kw in msg for kw in booking_kw):
                    score += 0.5
                if msg.strip():
                    score += 0.2
                scores.append(min(1.0, score))
            snap.mailbox_intent_score = round(sum(scores) / len(scores), 3)
    except Exception as e:
        log.warning(f"mailbox error: {e}")

    # Search rank
    try:
        results = api.search_masseurs(city="manhattan-ny", page=1)
        users = results.get("users", [])
        snap.available_rank = len(users)
        # Find our position (if username available)
        for i, user in enumerate(users):
            if hasattr(api, 'username') and user.get("username") == api.username:
                snap.search_rank = i + 1
                break
    except Exception as e:
        log.warning(f"search error: {e}")

    # Compute snapshot hash
    d = {k: v for k, v in asdict(snap).items() if k != "snapshot_hash"}
    snap.snapshot_hash = hashlib.sha256(
        json.dumps(d, sort_keys=True, default=str).encode()
    ).hexdigest()[:16]

    return snap

def get_baseline(conn: sqlite3.Connection, tenant_id: str, hours: int = 24) -> Optional[Dict]:
    """Get baseline metrics from the last N hours."""
    cur = conn.cursor()
    cur.execute("""
        SELECT AVG(views) as avg_views, AVG(contact_clicks) as avg_contacts,
               AVG(emails_count) as avg_emails, AVG(new_emails) as avg_new_emails,
               AVG(mailbox_intent_score) as avg_intent,
               AVG(booking_proxy_count) as avg_booking_proxy,
               AVG(repeat_sender_count) as avg_repeat,
               AVG(available_minutes) as avg_avail_min,
               COUNT(*) as n
        FROM metric_snapshots
        WHERE tenant_id = ? AND ts >= datetime('now', ?)
    """, (tenant_id, f'-{hours} hours'))
    row = cur.fetchone()
    if not row or row["n"] == 0:
        return None
    return dict(row)


def log_action(conn: sqlite3.Connection, tenant_id: str, action_type: str,
               endpoint: str = "", payload: str = "", error: str = "",
               receipt_hash: str = "") -> int:
    """Log an action to the database."""
    payload_hash = hashlib.sha256(payload.encode()).hexdigest()[:8] if payload else ""
    cur = conn.execute("""
        INSERT INTO actions (tenant_id, ts, action_type, payload_hash, endpoint_used, executed, error, receipt_hash)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """, (tenant_id, datetime.now(timezone.utc).isoformat(), action_type,
          payload_hash, endpoint, 1 if not error else 0, error, receipt_hash))
    conn.commit()
    return cur.lastrowid


def get_latest_snapshot(conn: sqlite3.Connection, tenant_id: str) -> Optional[Dict]:
    """Get the most recent snapshot for a tenant."""
    cur = conn.cursor()
    cur.execute("""
        SELECT * FROM metric_snapshots WHERE tenant_id = ? ORDER BY id DESC LIMIT 1
    """, (tenant_id,))
    row = cur.fetchone()
    return dict(row) if row else None

def dismiss_popups(driver):
    time.sleep(1)
    try:
        driver.execute_script("document.querySelectorAll('.DialogOverlay,[class*=\"overlay\"]').forEach(e=>e.remove())")
    except Exception:

def get_api_calls(driver):
    logs = driver.get_log("performance")
    reqs = {}
    for entry in logs:
        try:
            msg = json.loads(entry["message"])["message"]
            if msg["method"] == "Network.requestWillBeSent":
                p = msg["params"]
                r = p["request"]
                rid = p["requestId"]
                reqs[rid] = {
                    "url": r["url"],
                    "method": r["method"],
                    "postData": r.get("postData", "")[:2000],
                }
            elif msg["method"] == "Network.responseReceived":
                p = msg["params"]
                rid = p["requestId"]
                if rid in reqs:
                    reqs[rid]["status"] = p["response"]["status"]
                    try:
                        body = driver.execute_cdp_cmd("Network.getResponseBody", {"requestId": rid})
                        reqs[rid]["response"] = body.get("body", "")[:2000]
                    except Exception:
                        pass
        except Exception:
            pass
    return [v for v in reqs.values() if "/api/" in v["url"]]


def find_clickable(driver, texts):

    return None

def submit_draft(variant_id: str, kind: str, headline: str, description: str,
                 title: str = "", body: str = "", hypothesis: str = "") -> str:
    """Submit a draft to the approval queue."""
    upsert_content_variant(variant_id, kind, headline, description, title, body,
                         status="pending_approval", hypothesis=hypothesis)
    return variant_id


def approve_draft(variant_id: str) -> bool:
    """Mark a draft as approved."""
    set_variant_status(variant_id, "approved")
    return True


def reject_draft(variant_id: str) -> bool:
    """Reject a draft."""
    set_variant_status(variant_id, "rejected")
    return True


def list_pending(kind: str = None) -> list:
    """List pending drafts."""
    conn = get_conn()
    if kind:
        rows = conn.execute(
            "SELECT * FROM content_variants WHERE kind=? AND status='pending_approval' ORDER BY created_at DESC",
            (kind,)
        ).fetchall()
    else:
        rows = conn.execute(
            "SELECT * FROM content_variants WHERE status='pending_approval' ORDER BY created_at DESC"
        ).fetchall()
    conn.close()
    return [dict(r) for r in rows]
