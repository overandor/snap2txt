        snap.views = int(ps.get("totalPageViews", 0))
        snap.contact_clicks = int(ps.get("totalContactClicks", 0))
        snap.is_hidden = bool(dash.get("isAdHidden", False))
    except Exception as e:
        log.warning(f"dashboard error: {e}")

    # Ad statistics
    try:
        stats = api.get_ad_statistics()
        ps = stats.get("profileStatistics") or {}
        if not snap.views:
            snap.views = int(ps.get("totalPageViews", 0))
        if not snap.contact_clicks:
            snap.contact_clicks = int(ps.get("totalContactClicks", 0))
    except Exception as e:
        log.warning(f"ad_statistics error: {e}")

    # KeepOnline
    try:
        keep = api.get_keeponline()
        snap.new_emails = int(keep.get("newEmails", 0))
        snap.new_visits = int(keep.get("newVisits", 0))
        snap.is_hidden = snap.is_hidden or bool(keep.get("isAdHidden", False))
    except Exception as e:
        log.warning(f"keeponline error: {e}")

    # Availability
    try:
        avail = api.get_availability()
        selected = avail.get("selected", "")
        snap.is_available = (selected == "Available")
        countdown = int(avail.get("countdown", 0) or 0)
        if countdown:
            snap.available_minutes = max(0, (countdown - time.time()) / 60.0)
    except Exception as e:
        log.warning(f"availability error: {e}")

    # About
    try:
        about = api.get_about()
        props = about.get("userProps", {})
        assets = props.get("assets", {})
        headline = assets.get("headline", "")
        info = props.get("information", {})
        bio_text = info.get("description", "") or ""
        snap.headline_hash = hashlib.sha256(headline.encode()).hexdigest()[:8] if headline else ""
        snap.bio_hash = hashlib.sha256(bio_text.encode()).hexdigest()[:8] if bio_text else ""
    except Exception as e:
        log.warning(f"about error: {e}")

    # Mailbox
    try:
        mailbox = api.get_mailbox(page=1, folder=1, sort=1)
        emails = mailbox.get("emails", [])
        snap.emails_count = len(emails)

        booking_kw = ["book", "today", "tonight", "tomorrow", "session", "schedule", "available"]
        repeat_kw = ["again", "last time", "saw you", "been a while", "regular", "missed you"]

        for email in emails:
            msg = (email.get("lastMessage", "") or "").lower()
            if any(kw in msg for kw in booking_kw):
                snap.booking_proxy_count += 1
            if any(kw in msg for kw in repeat_kw):
                snap.repeat_sender_count += 1

        if emails:
            scores = []
            for email in emails:
                msg = (email.get("lastMessage", "") or "").lower()
                score = 0.0
                if any(kw in msg for kw in booking_kw):
                    score += 0.5
                if msg.strip():
                    score += 0.2
                scores.append(min(1.0, score))
            snap.mailbox_intent_score = round(sum(scores) / len(scores), 3)
    except Exception as e:
        log.warning(f"mailbox error: {e}")

    # Search rank
    try:
        results = api.search_masseurs(city="manhattan-ny", page=1)
        users = results.get("users", [])
        snap.available_rank = len(users)
        # Find our position (if username available)
        for i, user in enumerate(users):
            if hasattr(api, 'username') and user.get("username") == api.username:
                snap.search_rank = i + 1
                break
    except Exception as e:
        log.warning(f"search error: {e}")

    # Compute snapshot hash
    d = {k: v for k, v in asdict(snap).items() if k != "snapshot_hash"}
    snap.snapshot_hash = hashlib.sha256(
        json.dumps(d, sort_keys=True, default=str).encode()
    ).hexdigest()[:16]

    return snap

def get_baseline(conn: sqlite3.Connection, tenant_id: str, hours: int = 24) -> Optional[Dict]:
    """Get baseline metrics from the last N hours."""
    cur = conn.cursor()
    cur.execute("""
        SELECT AVG(views) as avg_views, AVG(contact_clicks) as avg_contacts,
               AVG(emails_count) as avg_emails, AVG(new_emails) as avg_new_emails,
               AVG(mailbox_intent_score) as avg_intent,
               AVG(booking_proxy_count) as avg_booking_proxy,
               AVG(repeat_sender_count) as avg_repeat,
               AVG(available_minutes) as avg_avail_min,
               COUNT(*) as n
        FROM metric_snapshots
        WHERE tenant_id = ? AND ts >= datetime('now', ?)
    """, (tenant_id, f'-{hours} hours'))
    row = cur.fetchone()
    if not row or row["n"] == 0:
        return None
    return dict(row)


def log_action(conn: sqlite3.Connection, tenant_id: str, action_type: str,
               endpoint: str = "", payload: str = "", error: str = "",
               receipt_hash: str = "") -> int:
    """Log an action to the database."""
    payload_hash = hashlib.sha256(payload.encode()).hexdigest()[:8] if payload else ""
    cur = conn.execute("""
        INSERT INTO actions (tenant_id, ts, action_type, payload_hash, endpoint_used, executed, error, receipt_hash)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """, (tenant_id, datetime.now(timezone.utc).isoformat(), action_type,
          payload_hash, endpoint, 1 if not error else 0, error, receipt_hash))
    conn.commit()
    return cur.lastrowid


def get_latest_snapshot(conn: sqlite3.Connection, tenant_id: str) -> Optional[Dict]:
    """Get the most recent snapshot for a tenant."""
    cur = conn.cursor()
    cur.execute("""
        SELECT * FROM metric_snapshots WHERE tenant_id = ? ORDER BY id DESC LIMIT 1
    """, (tenant_id,))
    row = cur.fetchone()
    return dict(row) if row else None

std::string replace_all(std::string str, const std::string& from, const std::string& to) {
    size_t start_pos = 0;
    while ((start_pos = str.find(from, start_pos)) != std::string::npos) {
        str.replace(start_pos, from.length(), to);
        start_pos += to.length();
    }
    return str;
}

std::string json_escape(const std::string& s) {
    std::string out;
    for (char c : s) {
        switch (c) {
            case '"': out += "\\\""; break;
            case '\\': out += "\\\\"; break;
            case '\n': out += "\\n"; break;
            case '\r': out += "\\r"; break;
            case '\t': out += "\\t"; break;
            default: out += c;
        }
    }
    return out;
}

"""
Deep token backend — unigrams, bigrams, trigrams, subword tokens, syllable tokens.
"""

import re
import json
from collections import Counter
from pathlib import Path
from typing import Dict, List, Tuple

from .bio_tokenizer import tokenize, _syllable_count


def extract_ngrams(tokens: List[str], n: int = 2) -> List[Tuple[str, ...]]:
    """Extract n-grams from token list."""
    return [tuple(tokens[i:i+n]) for i in range(len(tokens) - n + 1)]


def extract_subword_tokens(word: str, min_len: int = 3) -> List[str]:
    """Extract subword chunks: prefixes, suffixes, roots."""
    if len(word) <= min_len:
        return [word]
    chunks = []
    # Prefix 3, suffix 3, middle
    for i in range(0, len(word) - min_len + 1):
        chunks.append(word[i:i+min_len])
    return chunks


def extract_syllable_tokens(text: str) -> List[str]:
    """Extract syllable-count tokens per word."""
    words = tokenize(text)
    return [f"{w}:{_syllable_count(w)}" for w in words]


def extract_positional_tokens(text: str) -> List[str]:
    """Extract tokens with position: first, middle, last sentence."""
    sentences = re.split(r'[.!?]+', text)
    sentences = [s.strip() for s in sentences if s.strip()]
    result = []
    for i, sent in enumerate(sentences):
        words = tokenize(sent)
        for j, w in enumerate(words):
            if i == 0 and j == 0:
                result.append(f"FIRST:{w}")
            elif i == len(sentences) - 1 and j == len(words) - 1:
                result.append(f"LAST:{w}")
            else:
                result.append(f"MID:{w}")
    return result


def extract_structural_tokens(text: str) -> List[str]:
    """Extract structure signals: paragraphs, sentence count, punctuation."""
    paragraphs = len([p for p in text.split("\n") if p.strip()])
    sentences = len(re.split(r'[.!?]+', text))
    questions = text.count("?")
    exclamations = text.count("!")
    return [
        f"PARA:{paragraphs}",
        f"SENTS:{min(sentences, 10)}",
        f"Q:{questions}",
        f"EX:{exclamations}",
    ]


def deep_tokenize(headline: str, description: str) -> Dict:
    """Full token extraction."""
    text = headline + " " + description
    tokens = tokenize(text)
    return {
        "unigrams": tokens,
        "bigrams": [" ".join(bg) for bg in extract_ngrams(tokens, 2)],
        "trigrams": [" ".join(tg) for tg in extract_ngrams(tokens, 3)],
        "subwords": [sw for w in tokens for sw in extract_subword_tokens(w)],
        "syllables": extract_syllable_tokens(text),
        "positional": extract_positional_tokens(text),
        "structural": extract_structural_tokens(text),
    }
