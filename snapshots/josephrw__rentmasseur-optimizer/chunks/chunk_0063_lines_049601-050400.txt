
    model = MLP(input_size=n_features, hidden_size=48, output_size=3, seed=42, dropout=0.15)

    history = []
    best_val_loss = float("inf")
    best_model_state = None

    for epoch in range(epochs):
        # Train
        train_loss = model.train_epoch(X_train, y_train, lr=LEARNING_RATE, batch_size=BATCH_SIZE)

        # Validate
        y_pred = model.predict(X_val)
        val_loss = float(np.mean((y_pred - y_val) ** 2))
        val_mae = float(np.mean(np.abs(y_pred - y_val)))

        # Money reward: revenue signal + proof density - hallucination risk
        revenue_signal = float(np.mean(y_pred[:, 0]))  # avg predicted CTR
        proof_density = 1.0 - val_mae  # closer predictions = more proof
        hallucination_risk = float(np.mean(np.clip(y_pred - 0.15, 0, None)))  # over-prediction penalty
        money_reward = revenue_signal * 100 + proof_density * 50 - hallucination_risk * 20

        history.append({
            "epoch": epoch + 1,
            "train_loss": round(train_loss, 6),
            "val_loss": round(val_loss, 6),
            "val_mae": round(val_mae, 6),
            "revenue_signal": round(revenue_signal, 6),
            "money_reward": round(money_reward, 2),
        })

        if val_loss < best_val_loss:
            best_val_loss = val_loss
            best_model_state = {
                "W1": model.W1.copy(),
                "b1": model.b1.copy(),
                "W2": model.W2.copy(),
                "b2": model.b2.copy(),
            }

        if epoch % 5 == 0 or epoch == epochs - 1:
            log.info(
                "Epoch %2d/%d: train_loss=%.6f val_loss=%.6f mae=%.6f reward=%.2f",
                epoch + 1, epochs, train_loss, val_loss, val_mae, money_reward,
            )

    # Restore best model
    if best_model_state:
        model.W1 = best_model_state["W1"]
        model.b1 = best_model_state["b1"]
        model.W2 = best_model_state["W2"]
        model.b2 = best_model_state["b2"]

    # Save
    MODEL_PATH.parent.mkdir(parents=True, exist_ok=True)
    model.save(MODEL_PATH)
    log.info("Model saved to %s", MODEL_PATH)

    # Final metrics
    y_pred_val = model.predict(X_val)
    final_mae = float(np.mean(np.abs(y_pred_val - y_val)))
    final_rmse = float(np.sqrt(np.mean((y_pred_val - y_val) ** 2)))
    y_mean = np.mean(y_val, axis=0)
    ss_res = np.sum((y_val - y_pred_val) ** 2)
    ss_tot = np.sum((y_val - y_mean) ** 2)
    final_r2 = float(1 - ss_res / (ss_tot + 1e-10))

    summary = {
        "epochs": epochs,
        "train_size": split,
        "val_size": X.shape[0] - split,
        "features": n_features,
        "final_train_loss": round(history[-1]["train_loss"], 6),
        "final_val_loss": round(history[-1]["val_loss"], 6),
        "final_mae": round(final_mae, 6),
        "final_rmse": round(final_rmse, 6),
        "final_r2": round(final_r2, 4),
        "best_val_loss": round(best_val_loss, 6),
        "money_reward_final": history[-1]["money_reward"],
        "history": history,
    }

    return model, summary


def predict_top_bios(model: MLP, bios: List[Dict], top_n: int = 10) -> List[Dict]:
    """Use trained model to predict money-making potential of bios."""
    results = []
    for bio in bios:
        headline = bio.get("headline", "")
        description = bio.get("description", "")
        fv = feature_vector(headline, description)
        pred = model.predict(np.array([fv]))[0]
        results.append({
            "username": bio.get("username", "?"),
            "headline": headline[:60],
            "predicted_ctr": round(float(pred[0]), 6),
            "predicted_email": round(float(pred[1]), 6),
            "predicted_phone": round(float(pred[2]), 6),
            "money_score": round(float(pred[0] * 100 + pred[1] * 50 + pred[2] * 30), 4),
            "views": bio.get("views", 0),
        })
    results.sort(key=lambda x: x["money_score"], reverse=True)
    return results[:top_n]

def extract_bios_from_results(results: Dict, city: str = "") -> List[Dict]:
    """Extract bio data from search results."""
    bios = []
    users = results.get("users", [])
    if isinstance(users, dict):
        users = users.get("list", users.get("data", []))
    for user in users:
        if not isinstance(user, dict):
            continue
        headline = user.get("headline", "")
        description = user.get("description", "")
        if not description and not headline:
            continue
        card = user.get("userCard", {})
        bios.append({
            "id": card.get("userId", ""),
            "username": card.get("username", ""),
            "city": city or card.get("searchCity", card.get("location", "")),
            "headline": headline,
            "description": description,
            "isAvailable": card.get("status", {}).get("available", 0),
            "isGold": card.get("isGold", 0),

            "reviewsCount": user.get("reviewsCount", 0),
            "isCertified": user.get("isCertified", 0),
            "services": user.get("services", []),
            "distance": user.get("distance", 0),
            "travels": user.get("travels", []),
        })
    return bios

    for city in CITIES:
        log.info("Scraping %s...", city)
        for page in range(1, max_pages + 1):
            try:
                results = search_city(api, city, page)
                bios = extract_bios_from_results(results, city)
                if not bios:
                    break
                new_count = 0
                for bio in bios:
                    if bio["id"] and bio["id"] not in seen_ids:
                        seen_ids.add(bio["id"])
                        all_bios.append(bio)
                        new_count += 1
                log.info("  %s p%d: %d bios (%d new, total %d)", city, page, len(bios), new_count, len(all_bios))
                if new_count == 0:
                    break
                time.sleep(delay)
            except Exception as e:
                log.error("  Error on %s page %d: %s", city, page, e)
                break

    log.info("Scraped %d unique bios from %d cities", len(all_bios), len(CITIES))
    return all_bios


def save_bios(bios: List[Dict], path: Path = OUTPUT_PATH):
    """Save bios to JSONL."""
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w") as f:
        for bio in bios:
            f.write(json.dumps(bio, default=str) + "\n")
    log.info("Saved %d bios to %s", len(bios), path)


def rank_by_engagement(bios: List[Dict]) -> List[Dict]:
    """Rank bios by engagement proxy: reviews * rating."""
    def score(b):
        try:
            rating = float(b.get("ratingAverage", 0) or 0)
        except (ValueError, TypeError):
            rating = 0
        reviews = b.get("reviewsCount", 0) or 0
        return rating * reviews
    return sorted(bios, key=score, reverse=True)


def save_ranked_report(bios: List[Dict], path: Path = None):
    """Save a ranked report of all bios by engagement."""
    if path is None:
        path = Path(__file__).parent / "data" / "real_bios_ranked.txt"
    ranked = rank_by_engagement(bios)
    path.parent.mkdir(parents=True, exist_ok=True)
    with open(path, "w") as f:
        for i, b in enumerate(ranked):
            try:
                rating = float(b.get("ratingAverage", 0) or 0)
            except (ValueError, TypeError):
                rating = 0
            reviews = b.get("reviewsCount", 0) or 0
            engagement = rating * reviews
            f.write(f"#{i+1} | {b['username']} | {b['city']} | ⭐{rating} | {reviews} reviews | engagement={engagement}\n")
            f.write(f"  Headline: {b['headline']}\n")
            desc = b['description'].replace('\n', ' | ')[:500]
            f.write(f"  Desc: {desc}\n")
            f.write(f"  Services: {b.get('services', [])}\n")
            f.write(f"  Gold: {b.get('isGold', 0)} | Avail: {b.get('isAvailable', 0)} | Certified: {b.get('isCertified', 0)}\n")
            f.write("\n")
    log.info("Ranked report saved to %s", path)
    return path


def scrape_and_save(api: RentMasseurAPI, max_pages: int = 10, delay: float = 0.8) -> int:
    """Full scrape pipeline with ranking."""
    bios = scrape_all_cities(api, max_pages=max_pages, delay=delay)
    save_bios(bios)
    save_ranked_report(bios)
    return len(bios)
