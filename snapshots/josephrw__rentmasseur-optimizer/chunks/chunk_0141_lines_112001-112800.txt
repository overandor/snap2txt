    file_match = re.search(r'file\s*\(\s*id:\s*"([^"]+)"\s*\)', query)

        result = {}
        for f in fields:
            if f in ("filename",): result["filename"] = meta.get("filename", "unknown")
            elif f in ("size", "size_human"): result["size"] = meta.get("size_human", "?")
            elif f in ("format",): result["format"] = meta.get("format", "?")
            elif f in ("merkle", "merkle_root"): result["merkle_root"] = meta.get("merkle_root", "?")
            elif f in ("lines", "total_lines"): result["total_lines"] = meta.get("total_lines", "?")
            elif f in ("entropy",): result["entropy"] = meta.get("entropy", "?")
            elif f in ("engine", "engine_version"): result["engine_version"] = meta.get("engine_version", "1.0")
        return {"data": {"file": result}}
    # List all files
    if "files" in query:
        reg = load_registry()
        return {"data": {"files": [{"file_id": fid, "filename": info["filename"]} for fid, info in reg.items()]}}
    return {"errors": [{"message": "Unsupported query. Use: { file(id: \"xxx\") { filename size format } } or { files { filename } }"}]}


# --- Clipboard Superposition ---
# Multiple Jorki sessions encoded into a single base64 blob for clipboard.
# Format: JORKI:v1:<base64(json({sessions: [{file_id, filename, url, added_at}]}))>

SUPERPOSE_PREFIX = "JORKI:v1:"

def encode_superposition(file_ids: list) -> str:
    """Pack multiple file_ids into a single base64 clipboard string."""
    reg = load_registry()
    sessions = []
    for fid in file_ids:
        if fid in reg:
            info = reg[fid]
            sessions.append({
                "file_id": fid,
                "filename": info["meta"].get("filename", "unknown"),
                "size": info["meta"].get("size_human", "?"),
                "url": f"/meta/{fid}",
                "added_at": time.time(),
            })
    payload = json.dumps({"sessions": sessions, "created_at": time.time()}).encode()
    return SUPERPOSE_PREFIX + base64.b64encode(payload).decode()

def decode_superposition(blob: str) -> dict:
    """Unpack a superposition blob into session list."""
    blob = blob.strip()
    if not blob.startswith(SUPERPOSE_PREFIX):
        raise HTTPException(400, {"status": "invalid_blob", "message": "Not a Jorki superposition blob."})
    try:
        payload = base64.b64decode(blob[len(SUPERPOSE_PREFIX):])
        return json.loads(payload)
    except Exception:
        raise HTTPException(400, {"status": "invalid_blob", "message": "Corrupted superposition blob."})

@app.post("/superpose/decode")

    """Decode a superposition blob back into session list with live status."""
